    function renderGames(list) {

      const grid = document.getElementById('gameGrid');

      grid.innerHTML = list.map(g => `

        <div class="card">

          <div>

            <span class="badge ${g.platform}">${g.platform}</span>

            <h3>${g.name}</h3>

            <div class="genre">${g.tag}</div>

            <p>${g.desc}</p>

          </div>

          <a href="${g.url}" target="_blank" rel="noopener" class="play-btn">Play / View Game ➔</a>

        </div>

      `).join('');

    }


    function executeSearch() {

      const query = document.getElementById('searchInput').value.trim();

      const statusText = document.getElementById('statusText');


      if (!query) {

        statusText.innerText = "Top Featured Games";

        renderGames(initialGames);

        return;

      }


      statusText.innerText = `Searching results for "${query}"...`;


      fetch(`https://itunes.apple.com/search?term=${encodeURIComponent(query)}&entity=software&limit=6`)

        .then(res => res.json())

        .then(data => {

          const apiGames = data.results.map(item => ({

            name: item.trackName,

            platform: "appstore",

            tag: item.primaryGenreName || "App Store Game",

            desc: item.description ? item.description.substring(0, 90) + "..." : "Available on iOS App Store.",

            url: item.trackViewUrl

          }));


          const dynamicResults = [

            ...apiGames,

            {

              name: `${query} (Roblox)`,

              platform: "roblox",

              tag: "Roblox Experience",

              desc: `Find and play top Roblox experiences for "${query}".`,

              url: `https://www.roblox.com/discover#?keyword=${encodeURIComponent(query)}`

            },

            {

              name: `${query} on Poki`,

              platform: "web",

              tag: "Poki Games",

              desc: `Play free web browser games matching "${query}" on Poki.`,

              url: `https://poki.com/en/g/${encodeURIComponent(query.toLowerCase().replace(/\s+/g, '-'))}`

            },

            {

              name: `${query} on CrazyGames`,

              platform: "web",

              tag: "CrazyGames",

              desc: `Discover instantly playable HTML5 browser games on CrazyGames.`,

              url: `https://www.crazygames.com/search?q=${encodeURIComponent(query)}`

            }

          ];


          statusText.innerText = `Showing results for "${query}"`;

          renderGames(dynamicResults);

        })

        .catch(() => {

          statusText.innerText = `Search results for "${query}"`;

          renderGames([

            {

              name: `${query} (Roblox)`,

              platform: "roblox",

              tag: "Roblox Experience",

              desc: `Find Roblox games related to "${query}".`,

              url: `https://www.roblox.com/discover#?keyword=${encodeURIComponent(query)}`

            },

            {

              name: `${query} (App Store)`,

              platform: "appstore",

              tag: "Mobile App",

              desc: `Search App Store for "${query}".`,

              url: `https://www.apple.com/us/search/${encodeURIComponent(query)}`

            },

            {

              name: `${query} (CrazyGames)`,

              platform: "web",

              tag: "CrazyGames",

              desc: `Search CrazyGames for "${query}".`,

              url: `https://www.crazygames.com/search?q=${encodeURIComponent(query)}`

            }

          ]);

        });

    }


    const searchInput = document.getElementById('searchInput');

    const searchBtn = document.getElementById('searchBtn');


    // Fully isolates spacebar and input events so nothing disappears

    searchInput.addEventListener('input', function() {

      executeSearch();

    });


    searchBtn.addEventListener('click', function(e) {

      e.preventDefault();

      executeSearch();

    });


    renderGames(initialGames);
